  <h1>Relay Control</h1>
  <button onclick="sendCommand(1)">Turn ON</button>
  <button onclick="sendCommand(0)">Turn OFF</button>

  <script>
    function sendCommand(value) {
      const apiKey = 'Y9UZSEKQI52AY8TO';
      const url = `https://api.thingspeak.com/update?api_key=${apiKey}&field1=${value}`;

      fetch(url)
        .then(response => response.text())
        .then(data => {
          alert(value == 1 ? "Relay ON command sent" : "Relay OFF command sent");
        })
        .catch(error => {
          alert("Error sending command");
          console.error(error);
        });
    }
  </script>
